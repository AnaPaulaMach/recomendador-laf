% AUDITORIA COMPACTA PARA PRUEBA RAPIDA DEL MOTOR LAF
% Cadena representativa: sabor (positiva y negativa).
% No reemplaza la auditoria completa de todas las reglas.

% =====================================================================
% PROGRAMA LAF 1 - RANKING DE PRODUCTOS
% Sistema argumentativo de recomendacion de cafe
% Trabajo Final - Ana Paula Machado - UNSE
% =====================================================================

% Estrategia de balanceo: D - Contraccion hacia una media a priori
% Dominio: Cafe
% Productos analizados: 1

% Productos y rango de precios:
%   # 1 producto_auditoria_producto_de_auditoria       $100-$120  (2 publicaciones)

% Preferencias del usuario:
%   sabor: 0.50

% Estructura argumentativa LAF 1:
%   resena -> aspecto -> dimension -> recomendar
%   beta de reglas = 1.00 (neutral); la preferencia proviene solo del usuario
%   sensorial: sabor, aroma, intensidad, dulzor, tueste, acidez
%   valor: rendimiento, tamano
%   producto: originalidad, presentacion, variedad
%   elaboracion: preparacion, molienda
%   conservacion: caducidad

% ============================================================
% HECHOS - un hecho por (producto, aspecto, polaridad)
% Variable X identifica al producto.
% ============================================================

% Producto de auditoria
% n=3
resenaPositivaSabor(producto_auditoria_producto_de_auditoria). {0.65; 0.50}
% n=2
resenaNegativaSabor(producto_auditoria_producto_de_auditoria). {0.35; 0.50}

% ============================================================
% REGLAS DE AGREGACION POR ASPECTO
% ============================================================

saborBueno(X) :- resenaPositivaSabor(X). {0.85; 1.00}
~saborBueno(X) :- resenaNegativaSabor(X). {0.85; 1.00}

% ============================================================
% REGLAS INTERMEDIAS POR DIMENSION
% Cada dimension agrupa aspectos relacionados antes de recomendar.
% ============================================================

% Experiencia sensorial
sensorialBueno(X) :- saborBueno(X). {0.90; 1.00}
~sensorialBueno(X) :- ~saborBueno(X). {0.92; 1.00}

% ============================================================
% REGLAS DE RECOMENDACION
% X = producto candidato
% ============================================================

recomendar(X) :- sensorialBueno(X). {0.90; 1.00}
~recomendar(X) :- ~sensorialBueno(X). {0.92; 1.00}

% ============================================================
% ============================================================
% Atributo 1 - Confianza:
%   producto = X*Y     suma = X+Y-X*Y     debilitamiento = (X-Y)/(1-Y) si X>=Y, sino 0
% Atributo 2 - Preferencia del usuario:
%   producto = min(X,Y)     suma = min(X+Y,1)     debilitamiento = max(X-Y,0)
%
% Fuente: Budan et al. (2017) IJAR 82:242-269, Ejemplo 1 pp.249-250
